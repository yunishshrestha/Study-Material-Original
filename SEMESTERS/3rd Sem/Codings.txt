1. Bisection Method
#include<stdio.h>
#include<math.h>

#define f(x) (cos(x) - x * exp(x))

int main()
{
    float x0, x1, x2, f0, f1, f2, e;

up:
    printf("Enter two initial guesses:\n");
    scanf("%f%f", &x0, &x1);

    printf("Enter tolerable error:\n");
    scanf("%f", &e);

    f0 = f(x0);
    f1 = f(x1);

    if(f0 * f1 > 0.0)
    {
        printf("Incorrect Initial Guesses\n");
        goto up;
    }

    do
    {
        x2 = (x0 + x1)/2;
        f2 = f(x2);

        if(f0 * f2 < 0)
        {
            x1 = x2;
            f1 = f2;
        }
        else
        {
            x0 = x2;
            f0 = f2;
        }

    } while(fabs(f2) > e);

    printf("Root = %f", x2);
    return 0;
}


False Position Method
#include <stdio.h>
#include <math.h>

double f(double x)
{
    return pow(x, 3) - x - 1;
}

double false_position(double a, double b, double tol)
{
    double fa = f(a);
    double fb = f(b);
    double c, fc;

    do
    {
        c = (a*fb - b*fa)/(fb - fa);
        fc = f(c);

        if (fc * fa > 0)
        {
            a = c;
            fa = fc;
        }
        else
        {
            b = c;
            fb = fc;
        }

    } while (fabs(fc) > tol);

    return c;
}

int main()
{
    double x, y;

    printf("Enter initial guesses:\n");
    scanf("%lf%lf", &x, &y);

    if (f(x)*f(y) < 0)
    {
        double root = false_position(x, y, 0.0001);
        printf("Root = %f\n", root);
    }
    else
    {
        printf("Wrong guess\n");
    }

    return 0;
}

Newton Rhapson
#include <stdio.h>
#include <math.h>

#define EPSILON 0.000001

double f(double x)
{
    return x * log10(x) - 1.2;
}

double f_prime(double x)
{
    return 0.43429 + log10(x);
}

int main()
{
    double x;

    printf("Enter initial guess: ");
    scanf("%lf", &x);

    while(fabs(f(x)) > EPSILON)
    {
        x = x - f(x)/f_prime(x);
    }

    printf("Root = %lf", x);

}


Gauss Elimination (Partial Pivoting)
#include <stdio.h>
#include <math.h>

#define N 3

int main()
{
    float A[N][N+1], temp, ratio;
    int i,j,k,max_row;

    for(i=0;i<N;i++)
        for(j=0;j<=N;j++)
            scanf("%f",&A[i][j]);

    for(i=0;i<N;i++)
    {
        max_row=i;
        for(j=i+1;j<N;j++)
            if(fabs(A[j][i])>fabs(A[max_row][i]))
                max_row=j;

        for(k=0;k<=N;k++)
        {
            temp=A[i][k];
            A[i][k]=A[max_row][k];
            A[max_row][k]=temp;
        }

        for(j=i+1;j<N;j++)
        {
            ratio=A[j][i]/A[i][i];
            for(k=i;k<=N;k++)
                A[j][k]-=ratio*A[i][k];
        }
    }

    float x[N];
    for(i=N-1;i>=0;i--)
    {
        x[i]=A[i][N];
        for(j=i+1;j<N;j++)
            x[i]-=A[i][j]*x[j];
        x[i]/=A[i][i];
    }

    for(i=0;i<N;i++)
        printf("x%d=%f\n",i+1,x[i]);

    return 0;
}

Gauss Elimination (Normal)
#include <stdio.h>
#include <stdlib.h>

int main() {
    int i, j, k, n;
    printf("Enter the number of equations: ");
    scanf("%d", &n);

    double a[n][n+1], factor;

    printf("Enter the augmented matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j <= n; j++)
            scanf("%lf", &a[i][j]);

    // Forward elimination
    for (k = 0; k < n-1; k++) {
        for (i = k+1; i < n; i++) {
            factor = a[i][k] / a[k][k];
            for (j = k; j <= n; j++) {
                a[i][j] -= factor * a[k][j];
            }
        }
    }

    // Back substitution
    double x[n];
    for (i = n-1; i >= 0; i--) {
        x[i] = a[i][n];
        for (j = i+1; j < n; j++)
            x[i] -= a[i][j] * x[j];
        x[i] /= a[i][i];
    }

    printf("Solution:\n");
    for (i = 0; i < n; i++)
        printf("x[%d] = %lf\n", i, x[i]);

    return 0;
}




4. Linear Curve Fitting (y = a + bx)
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

void linear_fit(double x[], double y[], int n, double *a, double *b) {
    double sum_x = 0, sum_y = 0, sum_xy = 0, sum_xx = 0;
    int i;

    for (i = 0; i < n; i++) {
        sum_x += x[i];
        sum_y += y[i];
        sum_xy += x[i] * y[i];
        sum_xx += x[i] * x[i];
    }

    double delta = n * sum_xx - sum_x * sum_x;
    if (delta == 0) {
        printf("Cannot fit a line\n");
        exit(1);
    }

    *a = (n * sum_xy - sum_x * sum_y) / delta;
    *b = (sum_y - (*a) * sum_x) / n;
}

int main() {
    int n;
    printf("Enter the number of data points: ");
    scanf("%d", &n);

    double x[n], y[n];
    for (int i = 0; i < n; i++) {
        printf("Enter x[%d] and y[%d]: ", i, i);
        scanf("%lf %lf", &x[i], &y[i]);
    }

    double a, b;
    linear_fit(x, y, n, &a, &b);
    printf("a = %f, b = %f\n", a, b);
    printf("The best curve fit is y = %f + %f x\n", a, b);

    return 0;
}


5. Trapezoidal Rule
#include <stdio.h>
#include <math.h>

double function(double x) {
    return 4 * exp(x) / (1 + pow(x, 3)); // Replace with your function
}

double trapezoidalRule(double a, double b, int n) {
    double h = (b - a) / n;
    double sum = (function(a) + function(b)) / 2.0;

    for (int i = 1; i < n; i++) {
        double x = a + i * h;
        sum += function(x);
    }

    return h * sum;
}

int main() {
    double a, b;
    int n;
    printf("Enter lower limit a, upper limit b and number of subintervals n: ");
    scanf("%lf %lf %d", &a, &b, &n);

    double result = trapezoidalRule(a, b, n);
    printf("Numerical integration result: %lf\n", result);

    return 0;
}


6. Simpson 1/3 Rule
#include <stdio.h>
#include <math.h>

double function(double x) {
    return cos(x); // Replace with your function
}

double simpsonsRule(double a, double b, int n) {
    if (n % 2 != 0) {
        printf("Number of subintervals must be even.\n");
        return 0;
    }

    double h = (b - a) / n;
    double sum = function(a) + function(b);

    for (int i = 1; i < n; i++) {
        double x = a + i * h;
        sum += (i % 2 == 0) ? 2 * function(x) : 4 * function(x);
    }

    return (h / 3.0) * sum;
}

int main() {
    double a, b;
    int n;
    printf("Enter lower limit a, upper limit b and even number of subintervals n: ");
    scanf("%lf %lf %d", &a, &b, &n);

    double result = simpsonsRule(a, b, n);
    printf("Numerical integration result: %lf\n", result);

    return 0;
}

7. Gaussian Integration (2 and 3 points)
#include <stdio.h>

    return x * x; // Replace with your function

double gaussian_integration(double a, double b, int n) {
    double result = 0.0;

    if (n == 2) {
        double w[2] = {1.0, 1.0};
        double x[2] = {-0.5773502692, 0.5773502692};

        for (int i = 0; i < 2; i++) {
            double t = ((b - a) / 2.0) * x[i] + ((b + a) / 2.0);
            result += w[i] * f(t);
        }
    } else if (n == 3) {
        double w[3] = {0.5555555556, 0.8888888889, 0.5555555556};
        double x[3] = {-0.7745966692, 0.0, 0.7745966692};

        for (int i = 0; i < 3; i++) {
            double t = ((b - a) / 2.0) * x[i] + ((b + a) / 2.0);
            result += w[i] * f(t);
        }
    }

    return result * (b - a) / 2.0;
}

void rk4Method(double x0, double y0, double h, double xn) {
    double x = x0, y = y0;

    printf("x\t\t y\n");
    while (x <= xn) {
        printf("%lf\t %lf\n", x, y);
        double k1 = h * f(x, y);
        double k2 = h * f(x + h/2.0, y + k1/2.0);
        double k3 = h * f(x + h/2.0, y + k2/2.0);
        double k4 = h * f(x + h, y + k3);

    }
}

        y = y + h*y1 + (pow(h,2)/2)*y2 + (pow(h,3)/6)*y3 + (pow(h,4)/24)*y4;

        printf("%d\t%.2lf\t%.6lf\n", i, x, y); // print y_n at each step
    }

 1. Shooting Method (Boundary Value Problem)
#include <stdio.h>
#include <math.h>

#define m1 2.0
#define m2 4.0

double f(double x, double y) {
    return x * y;
}

double f1(double fp) {
    return 1 + fp;
}

double f2(double x, double y) {
    return x * y;
}

double taylor(double x0, double y0, double xn, double fp0) {
    double h = xn - x0;

    double y = y0
        + h * fp0
        + pow(h, 2) * f(x0, y0) / 2.0
        + pow(h, 3) * f1(fp0) / 6.0
        + pow(h, 4) * f2(x0, y0) / 24.0;

    return y;
}

int main() {
    double x0, y0, xn, yn;

    printf("Enter x0 and y0: ");
    scanf("%lf %lf", &x0, &y0);

    printf("Enter xn and yn: ");
    scanf("%lf %lf", &xn, &yn);

    double b1 = taylor(x0, y0, xn, m1);
    double b2 = taylor(x0, y0, xn, m2);

    double m = m2 - ((b2 - yn) / (b2 - b1)) * (m2 - m1);

    printf("\nf(xn) = %lf\n", b1);
    printf("\nf(xn) = %lf\n", b2);

    printf("\nEnter xn for which y is to be found: ");
    scanf("%lf", &xn);

    double res = taylor(x0, y0, xn, m);

    printf("y(%lf) = %lf\n", xn, res);

    return 0;
}

3. Newton Forward Interpolation
#include <stdio.h>

int factorial(int n) {
    int fact = 1;
    for (int i = 1; i <= n; i++) {
        fact *= i;
    }
    return fact;
}

float newtonForwardInterpolation(int n, float x[], float y[], float x0) {
    float y0 = y[0], u = (x0 - x[0]) / (x[1] - x[0]);
    float forward[n-1][n-1];

    for (int j = 0; j < n-1; j++) {
        for (int i = 0; i < n-j-1; i++) {
            if (j == 0) {
                forward[i][j] = y[i+1] - y[i];
            } else {
                forward[i][j] = forward[i+1][j-1] - forward[i][j-1];
            }
        }
    }

    printf("Forward Difference Table:\n");
    for (int i = 0; i < n-1; i++) {
        printf("%.4f\t", forward[i][0]);
        for (int j = 1; j < n-i-1; j++)
            printf("%.4f\t", forward[i][j]);
        printf("\n");
    }

    for (int j = 1; j < n; j++) {
        y0 += (u * forward[0][j-1]) / factorial(j);
        u *= (x0 - x[j]) / (x[j+1] - x[j]);
    }

    return y0;
}

int main() {
    int n;
    printf("Enter the number of data points: ");
    scanf("%d", &n);

    float x[n], y[n];
    for (int i = 0; i < n; i++) {
        printf("Enter x[%d] and y[%d]: ", i, i);
        scanf("%f %f", &x[i], &y[i]);
    }

    float x0;
    printf("Enter the value of x for which y is to be found: ");
    scanf("%f", &x0);

    float y0 = newtonForwardInterpolation(n, x, y, x0);
    printf("The value of y at x = %.2f is %.2f\n", x0, y0);

    return 0;
}






13. Backward Difference
#include <stdio.h>

double x[] = {1.0, 1.1, 1.2, 1.3, 1.4, 1.5, 1.6};
double y[] = {7.989, 8.403, 8.781, 9.129, 9.451, 9.750, 10.031};

double backwardDifference(int index) {
    double h = x[index] - x[index - 1];
    return (3*y[index] - 4*y[index - 1] + y[index - 2]) / (2*h);
}

int main() {
    int index;
    printf("Enter index: ");
    scanf("%d", &index);

    double result = backwardDifference(index);

8. Forward Difference for Tabulated Data
#include <stdio.h>
#include <math.h>

double x[] = {1.0, 1.1, 1.2, 1.3, 1.4, 1.5, 1.6};
double y[] = {7.989, 8.403, 8.781, 9.129, 9.451, 9.750, 10.031};

double forwardDifference(int index) {
    double h = x[index + 1] - x[index];
    return (-3*y[index] + 4*y[index + 1] - y[index + 2]) / (2*h);
}

int main() {
    int index;
    printf("Enter index at which derivative is required: ");
    scanf("%d", &index);

    double result = forwardDifference(index);
    printf("Numerical differentiation result: %.4lf\n", result);

    return 0;
}


Lagrange Interpolation
#include <stdio.h>

double lagrange_interp(double x[], double y[], int n, double xi) {
    double L, P = 1, result = 0;
    int i, j;

    for (i = 0; i < n; i++) {
        P = 1;
        for (j = 0; j < n; j++) {
            if (j != i) {
                P *= (xi - x[j]) / (x[i] - x[j]);
            }
        }
        L = y[i] * P;
        result += L;
    }
    return result;
}

int main() {
    int n, i;
    double xi;
    printf("Enter the number of data points: ");
    scanf("%d", &n);

    double x[n], y[n];
    for (i = 0; i < n; i++) {
        printf("Enter x[%d] and y[%d]: ", i, i);
        scanf("%lf %lf", &x[i], &y[i]);
    }

    printf("Enter the value to interpolate: ");
    scanf("%lf", &xi);

    double result = lagrange_interp(x, y, n, xi);
    printf("Interpolated value at %f is %f\n", xi, result);
    return 0;
}
